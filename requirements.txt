# Kafka - Use Confluent Kafka API (not kafka-python)
confluent-kafka==2.3.0

# Redis - job state store on the master node
redis==5.0.1

# MinIO - S3-compatible blob store for tile bytes (claim-check pattern)
minio==7.2.7

# Prometheus - metrics exposed by master and worker
prometheus-client==0.20.0

# Image Processing - headless only. Both opencv-python and
# opencv-python-headless install the same top-level `cv2` package; pinning
# both is a real conflict (whichever installs last "wins", and both being
# listed pulls in opencv-python's unneeded GUI/X11 bindings on top of it in
# a server process that never opens a window), found during a full-project
# review. This service is server-side only.
opencv-python-headless==4.8.1.78
numpy==1.26.2

# ML inference (Step 9) - plain CPU onnxruntime here, not onnxruntime-gpu.
# The GPU build (+ its CUDA 13 nvidia-* runtime libs, and the sm_120/PTX-JIT
# reasoning for pinning that exact CUDA 13 build) lives in
# requirements-gpu.txt instead, installed only when the Dockerfile's
# INSTALL_GPU build arg is set (see docker-compose.gpu.yml) - found during a
# full-project review: onnxruntime-gpu plus seven nvidia-* CUDA wheels were
# unconditional deps here, so `docker compose build` pulled multi-GB CUDA
# packages into the master/worker images too (which never touch CUDA), and
# failed outright on arm64 (those wheels are linux/x86_64-only, so a Mac
# Apple Silicon host - which the README's quickstart claims works on "any
# machine" - couldn't build at all). inference/engine.py's make_session()
# already requests CUDAExecutionProvider first and falls back to
# CPUExecutionProvider when it's unavailable, so a plain onnxruntime install
# here is a real fallback path, not a stub.
onnxruntime==1.20.1

# Benchmark harness (bench/) - runs on the host, not in a container, so it
# needs its own HTTP client and plotting; reuses this same venv rather than
# maintaining a second requirements file.
requests==2.31.0
matplotlib==3.8.4

# Tests (tests/) - host-only, same reasoning as bench/ above. testcontainers
# spins up a real Redis in Docker for tests/test_jobstore.py rather than a
# pure-Python Redis reimplementation, since what's under test is real Redis
# atomicity semantics (HSET-return-value gating, HLEN as source of truth) -
# see tests/conftest.py.
pytest==8.3.3
testcontainers[redis]==4.8.1
